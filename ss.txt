// Colors
const heartColors = ['#f5d0d0', '#d4a5a5', '#e07a5f', '#f2a794', '#ffd93d'];

// Create floating hearts
function createHeart() {
  const heart = document.createElement('div');
  heart.className = 'heart';

  const size = Math.random() * 20 + 10;
  const color = heartColors[Math.floor(Math.random() * heartColors.length)];
  const left = Math.random() * 100;
  const duration = Math.random() * 5 + 8;
  const delay = Math.random() * 3;

  heart.innerHTML = `
    <svg width="${size}" height="${size}" viewBox="0 0 30 30">
      <path d="M 15 5 c 0 -6 7 -10 12 -5 c 5 -5 12 -1 12 5 c 0 9 -12 18 -12 18 c 0 0 -12 -9 -12 -18" fill="${color}"/>
    </svg>
  `;

  heart.style.left = `${left}%`;
  heart.style.animationDuration = `${duration}s`;
  heart.style.animationDelay = `${delay}s`;

  heartsContainer.appendChild(heart);

  setTimeout(() => {
    if (heart.parentNode) heart.remove();
  }, (duration + delay) * 1000);
}

// Create confetti burst
function createConfetti() {
  const colors = ['#e07a5f', '#d4a5a5', '#ffd93d', '#7d9a6a', '#a58bc9'];
  const shapes = ['circle', 'square', 'triangle'];

  for (let i = 0; i < 80; i++) {
    const confetti = document.createElement('div');
    confetti.className = 'confetti';

    const color = colors[Math.floor(Math.random() * colors.length)];
    const shape = shapes[Math.floor(Math.random() * shapes.length)];
    const left = Math.random() * 100;
    const duration = Math.random() * 3 + 2;
    const size = Math.random() * 10 + 5;

    if (shape === 'circle') {
      confetti.style.borderRadius = '50%';
    } else if (shape === 'triangle') {
      confetti.style.width = '0';
      confetti.style.height = '0';
      confetti.style.borderLeft = `${size}px solid transparent`;
      confetti.style.borderRight = `${size}px solid transparent`;
      confetti.style.borderBottom = `${size * 1.5}px solid ${color}`;
      confetti.style.backgroundColor = 'transparent';
    }

    confetti.style.left = `${left}%`;
    confetti.style.backgroundColor = shape !== 'triangle' ? color : 'transparent';
    confetti.style.width = shape !== 'triangle' ? `${size}px` : confetti.style.width;
    confetti.style.height = shape !== 'triangle' ? `${size}px` : confetti.style.height;
    confetti.style.animationDuration = `${duration}s`;
    confetti.style.animationDelay = `${Math.random() * 0.5}s`;

    document.body.appendChild(confetti);

    setTimeout(() => {
      if (confetti.parentNode) confetti.remove();
    }, (duration + 0.5) * 1000);
  }
}

// Create sparkles
function createSparkles() {
  const card = document.querySelector('.message-card');
  if (!card) return;

  for (let i = 0; i < 12; i++) {
    const sparkle = document.createElement('div');
    sparkle.className = 'sparkle';
    sparkle.style.left = `${Math.random() * 100}%`;
    sparkle.style.top = `${Math.random() * 100}%`;
    sparkle.style.animationDelay = `${Math.random() * 2}s`;
    card.appendChild(sparkle);
  }
}

// Modal functions
function openModal() {
  specialModal.classList.add('active');
  setTimeout(() => {
    modalContent.classList.add('active');
  }, 50);
  createConfetti();
}

function closeModalFn() {
  modalContent.classList.remove('active');
  setTimeout(() => {
    specialModal.classList.remove('active');
  }, 300);
}

// Event listeners
celebrateBtn.addEventListener('click', openModal);
closeModal.addEventListener('click', closeModalFn);
modalOverlay.addEventListener('click', closeModalFn);

// Keyboard accessibility
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape' && specialModal.classList.contains('active')) {
    closeModalFn();
  }
});

// Initialize
function init() {
  // Create initial hearts
  for (let i = 0; i < 8; i++) {
    setTimeout(createHeart, i * 400);
  }

  // Continue creating hearts
  setInterval(createHeart, 1500);

  // Create sparkles
  setTimeout(createSparkles, 1000);

  // Initial confetti on load
  setTimeout(createConfetti, 500);
}

// Run init
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', init);
} else {
  init();
}
